#!/usr/bin/env bash
set -uo pipefail

echo "=== Alien Intelligence (AI²) Build & Test ==="

# Configure
cmake -B build -S . -DCMAKE_BUILD_TYPE=Release

# Build
cmake --build build -j"$(nproc)"

# Run all tests
echo ""
echo "=== Running Tests ==="
FAILED=0
for test in build/test_*; do
    if [ -x "$test" ]; then
        if "$test"; then
            :
        else
            FAILED=$((FAILED + 1))
        fi
        echo ""
    fi
done

if [ "$FAILED" -eq 0 ]; then
    echo "=== All components built and tested successfully ==="
else
    echo "=== $FAILED test suite(s) FAILED ==="
    exit 1
fi
