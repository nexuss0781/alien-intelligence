#!/usr/bin/env bash
set -uo pipefail

echo "=== Alien Intelligence (AI²) Training ==="

# Step 1: Download datasets
echo "[1/3] Downloading datasets..."
bash scripts/download_datasets.sh

# Step 2: Build training binary
echo "[2/3] Building..."
cmake -B build -S . -DCMAKE_BUILD_TYPE=Release
cmake --build build -j"$(nproc)" --target ai2_train

# Step 3: Run training
# Total steps: ~7638 (3 epochs × ~2546 batches/epoch with batch_size=8, seq_len=64)
echo "[3/3] Starting training (3 epochs ~ 7638 steps) ..."
./build/ai2_train
