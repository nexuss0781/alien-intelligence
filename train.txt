#!/usr/bin/env bash
set -uo pipefail

echo "=== Alien Intelligence (AI²) Training ==="

# Step 1: Download datasets
echo "[1/3] Downloading datasets (if needed)..."
if [ ! -f "data/pretrain.txt" ]; then
    bash scripts/download_datasets.sh
else
    echo "  Data already exists, skipping download"
fi

# Step 2: Build training binary with CUDA
echo "[2/3] Building with CUDA..."
cmake -B build -S . -DCMAKE_BUILD_TYPE=Release
cmake --build build -j"$(nproc)" --target ai2_train

# Step 3: Run training
echo "[3/3] Starting training (GPU) ..."
mkdir -p checkpoints model

# Prefix-aware resume: if finetune checkpoints exist, binary skips pretrain
if ls checkpoints/ai2_finetune_*.bin 2>/dev/null | head -1 | grep -q .; then
    echo "  Finetune checkpoint(s) found, resuming..."
    ./build/ai2_train --resume
elif ls checkpoints/ai2_pretrain_*.bin 2>/dev/null | head -1 | grep -q .; then
    echo "  Pretrain checkpoint(s) found, resuming..."
    ./build/ai2_train --resume
elif [ -f "model/model.bin" ]; then
    echo "  model/model.bin found, resuming pretrain..."
    ./build/ai2_train --resume
else
    echo "  No checkpoint found, starting fresh..."
    ./build/ai2_train
fi

echo ""
echo "=== Done ==="
echo "  Latest model: model/model.bin"
echo "  Step checkpoints: checkpoints/*_step_*.bin (every 100 steps)"
echo "  Epoch checkpoints: checkpoints/*_epoch_*.bin (every 3 epochs)"
