#!/usr/bin/env bash
set -uo pipefail

echo "=== Alien Intelligence (AI²) Training ==="

# Step 1: Download datasets
echo "[1/3] Downloading datasets (if needed)..."
if [ ! -f "data/pretrain.txt" ]; then
    bash scripts/download_datasets.sh
else
    echo "  Data already exists, skipping download"
fi

# Step 2: Build training binary
echo "[2/3] Building..."
cmake -B build -S . -DCMAKE_BUILD_TYPE=Release
cmake --build build -j"$(nproc)" --target ai2_train

# Step 3: Run training (use all available CPU cores)
export OMP_NUM_THREADS="${OMP_NUM_THREADS:-$(nproc)}"
echo "[3/3] Starting training (OMP_NUM_THREADS=$OMP_NUM_THREADS) ..."
./build/ai2_train
